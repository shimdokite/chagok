"use client";
import { useEffect } from "react";
import { requestId } from "../apply/request-id";
import { bridgeBase } from "@chagok/contracts";
import type { FormClient } from "../apply/form-client";

declare global {
  interface Window {
    ReactNativeWebView?: { postMessage: (data: string) => void };
    __CHAGOK_BRIDGE__?: { receive: (value: unknown) => void };
  }
}

export function BridgeReceiver({ client }: { client: FormClient }) {
  useEffect(() => {
    const receiver = { receive: (value: unknown) => client.receive(value) };

    const connect = () => {
      window.__CHAGOK_BRIDGE__ = receiver;
      client.start();
      if (!window.ReactNativeWebView)
        client.receive({
          ...bridgeBase,
          type: "FORM_INIT",
          connectionId: requestId(),
          payload: { mode: "demo" },
        });
    };

    const resume = (event: PageTransitionEvent) => {
      if (event.persisted) connect();
    };

    connect();
    // Temporary emulator-only direct child-frame message probe.
    const proof = document.createElement("p");
    proof.id = "frame-probe-proof";
    proof.textContent = "frame-probe waiting";
    document.body.prepend(proof);
    const frames: HTMLIFrameElement[] = [];
    for (const kind of ["srcdoc", "blank"]) {
      const frame = document.createElement("iframe");
      const payload = JSON.stringify(JSON.stringify({ ...bridgeBase, type: "CLOSE_WEBVIEW" }));
      const html = `<script>setTimeout(() => { const available = typeof window.ReactNativeWebView?.postMessage === 'function'; if (available) window.ReactNativeWebView.postMessage(${payload}); parent.document.getElementById('frame-probe-proof').textContent += ' / ${kind} posted=' + available; }, 2500);</script>`;
      if (kind === "srcdoc") frame.srcdoc = html;
      document.body.appendChild(frame);
      if (kind === "blank") { frame.contentDocument?.open(); frame.contentDocument?.write(html); frame.contentDocument?.close(); }
      frames.push(frame);
    }

    const cleanup = () => {
      frames.forEach((frame) => frame.remove());
      proof.remove();
      client.dispose();
      if (window.__CHAGOK_BRIDGE__ === receiver)
        delete window.__CHAGOK_BRIDGE__;
    };

    window.addEventListener("pagehide", cleanup);
    window.addEventListener("pageshow", resume);
    return () => {
      window.removeEventListener("pagehide", cleanup);
      window.removeEventListener("pageshow", resume);
      cleanup();
    };
  }, [client]);
  return null;
}
